Inheritance example :

1)
#include<iostream>
Using namespace std;
          class abc{
                public:
                int parentNum;
          };

          class bcd: public : abc{
                public:
                int childNum;
    
          int main(){ 
               bcd obj;
               obj.parentNum = 10;
               obj.childNum = 20;
               cout<<"parent number is :"<<obj.parentNum<<endl;
               cout<<"child number is :"<<obj.childNum<<endl;
               return 0;
         }

2)
#include<iostream>
using namespace std;
    class parent{
        public:
        int parentNum;
    };
    class child : public parent{
        public:
        int childNum;
    };
    class grandChild : public child{
        public:
            int grandChildNum;
              void display(){
                 grandChildNum = 30;
                 cout<<"number is : "<<grandChildNum<<endl;
        }
    };

    int main(){
        grandChild obj;
        obj.parentNum = 20;
        return 0;
    }
